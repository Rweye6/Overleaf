	\begin{enumerate}
		\item Calcul de l'économie d'atomes pour la réaction de Jones :
	\end{enumerate}
	Bilan des atomes de la réaction de Jones : \\
	Masse molaire de \ce{C8H10O} : $M(\ce{C8H10O}) = \SI{122}{\gram\per\mole}$\\
	Masse molaire de \ce{CrO3} : $M(\ce{CrO3}) = \SI{100}{\gram\per\mole}$\\
	Masse molaire de \ce{H2SO4} : $M(\ce{H2SO4}) = \SI{98,1}{\gram\per\mole}$\\
	Masse molaire de \ce{C8H8O} : $M(\ce{C8H8O}) = \SI{120}{\gram\per\mole}$\\
	Masse molaire de \ce{Cr2(SO4)3} : $M(\ce{Cr2(SO4)3}) = \SI{392,3}{\gram\per\mole}$\\
	Masse molaire de \ce{H2O} : $M(\ce{H2O}) = \SI{18}{\gram\per\mole}$
	$$ EA = \dfrac{3\times M(\ce{C8H8O}) + M(\ce{Cr2(SO4)3}) + 6\times M(\ce{H2O})}{3\times M(\ce{C8H10O}) + 2\times M(\ce{CrO3})+ 3 \times M(\ce{H2SO4})}$$

	$$ EA = \dfrac{3\times 120 + 392,3 + 6\times 18 }{3\times 122 + 2\times 100+ 3 \times 98,1}$$

	$$ EA = 1,006741834$$
